#define STRICT
#define WIN32_LEAN_AND_MEAN

#include <Windows.h>

LRESULT CALLBACK WndProc(HWND hWnd, UINT message, WPARAM wParam, LPARAM lParam);

int WINAPI WinMain(
    _In_ HINSTANCE hInstance,
    _In_opt_ HINSTANCE,
    _In_ LPSTR,
    _In_ int nCmdShow
)
{
    LPCTSTR szClass = TEXT("MyWindowClass");

    WNDCLASS wndclass = {};
    wndclass.lpfnWndProc = WndProc;
    wndclass.hInstance = hInstance;
    wndclass.lpszClassName = szClass;
    wndclass.hbrBackground = (HBRUSH)(COLOR_WINDOW + 1);

    RegisterClass(&wndclass);

    // Получаем размеры экрана и вычисляем координаты левого верхнего угла так, чтобы окно было по центру
    int screenW = ::GetSystemMetrics(SM_CXSCREEN);
    int screenH = ::GetSystemMetrics(SM_CYSCREEN);
    int winW = screenW * 0.5;
    int winH = screenH * 0.5;
    int posX = (screenW - winW) / 2;
    int posY = (screenH - winH) / 2;

    // Создаём главное окно
    HWND hWnd = ::CreateWindow(szClass, TEXT("Test - Rectangles"), WS_OVERLAPPEDWINDOW,
        posX, posY, winW, winH, NULL, NULL, hInstance, NULL);

    if (hWnd == NULL) {
        return -1;
    }

    ::ShowWindow(hWnd, nCmdShow);
    // ::UpdateWindow(hWnd);

    // Цикл сообщений. GetMessage возвращает 0, когда пришло WM_QUIT
    MSG msg;
    while (::GetMessage(&msg, NULL, 0, 0)) {
        // ::TranslateMessage(&msg);
        ::DispatchMessage(&msg); // отправляем сообщение в WndProc
    }

    return 0;
}

void quantityOfRectangles(int* pCols, int* pRows)
{
    const int N = 9;
    switch (N) {
    case 4:  *pCols = 2; *pRows = 2; break;
    case 6:  *pCols = 3; *pRows = 2; break;
    case 9:  *pCols = 3; *pRows = 3; break;
    default: *pCols = 2; *pRows = 2; break;
    }
    // это тоже неэффективно
}

// Возвращает прямоугольник клетки с номером index.
// Клетки нумеруются слева направо, сверху вниз: для 3x3 это 0 1 2 / 3 4 5 / 6 7 8.
RECT GetCellRect(HWND hWnd, int index)
{
    RECT clientRect;
    ::GetClientRect(hWnd, &clientRect); // left = 0 и top = 0

    int cols, rows;
    quantityOfRectangles(&cols, &rows);

    // Вычисляем ширину и высоту одной ячейки
    // Значение right равно точной ширине области в пикселях, а bottom — её высоте
    int cellW = clientRect.right / cols;
    int cellH = clientRect.bottom / rows;

    int col = index % cols; // остаток от деления - номер столбца
    int row = index / cols; // целая часть - номер строки

    RECT rect;
    rect.left = col * cellW; // сдвиг по горизонтали
    rect.top = row * cellH;  // сдвиг по вертикали
    // Если это последняя колонка, то правый край совпадает с правым краем клиентской области
    rect.right = (col == cols - 1) ? clientRect.right : (col + 1) * cellW;
    rect.bottom = (row == rows - 1) ? clientRect.bottom : (row + 1) * cellH;
    return rect; // так возвращать неэффективно(!)
}

// Возвращает номер клетки, в которой лежит точка pt. Если точка вне клеток, возвращает -1.
int HoverIndex(HWND hWnd, POINT pt)
{
    int cols, rows;
    quantityOfRectangles(&cols, &rows);

    // Определяем, в каком прямоугольнике находится курсор: перебираем клетки по номерам
    for (int i = 0; i < cols * rows; ++i) {
        RECT rc = GetCellRect(hWnd, i);
        if (::PtInRect(&rc, pt)) { // возвращает TRUE, если точка pt лежит внутри rc
            return i;              // нашли - возвращаем номер клетки
        }
    }
    return -1;
}

LRESULT CALLBACK WndProc(HWND hWnd, UINT message, WPARAM wParam, LPARAM lParam)
{
    switch (message) {

    case WM_CREATE:
        // Окно только что создано. Сохраняем в "USERDATA" окна значение -1: "курсор ни над одной клеткой"
        ::SetWindowLongPtr(hWnd, GWLP_USERDATA, (LONG_PTR)-1);
        return 0;

    case WM_SIZE:
    {
        // Размер окна изменился: старая сетка и подсветка уже не подходят.
        // Сбрасываем подсветку (она вернётся при первом же движении мыши) и просим перерисовать всё окно.
        ::SetWindowLongPtr(hWnd, GWLP_USERDATA, (LONG_PTR)-1);
        ::InvalidateRect(hWnd, NULL, FALSE); // окно; область(вся); bErase - не позволяет стирать всё окно, "новый ватман"
        return 0;
    }

    case WM_MOUSEMOVE:
    {
        POINT point;
        point.x = (short)LOWORD(lParam); // Извлекает младшие 16 бит
        point.y = (short)HIWORD(lParam);
        // Приведение к short необходимо, чтобы корректно обрабатывать отрицательные значения при выходе курсора за пределы окна
        // так как в противном случае превратится в большое число.

        int oldIndex = (int)::GetWindowLongPtr(hWnd, GWLP_USERDATA); // клетка, подсвеченная сейчас
        int newIndex = HoverIndex(hWnd, point);                   // клетка под курсором теперь

        // Если прямоугольник под мышью изменился, запрашиваем перерисовку.
        // Так окно не перерисовывается десятки раз в секунду, пока курсор двигается внутри одного и того же прямоугольника.
        if (oldIndex != newIndex) {
            ::SetWindowLongPtr(hWnd, GWLP_USERDATA, (LONG_PTR)newIndex); // запоминаем новую клетку
            // Сама отрисовка происходит только в WM_PAINT. FALSE: фон не стираем отдельно, WM_PAINT всё равно закрасит его белым.
            ::InvalidateRect(hWnd, NULL, FALSE);
        }
        return 0;
    }

    case WM_NCMOUSEMOVE:
    {
        // Когда мышь переходит на неклиентскую область (заголовок, рамка), сбрасываем выделение и восстанавливаем фон.
        int oldIndex = (int)::GetWindowLongPtr(hWnd, GWLP_USERDATA);
        if (oldIndex != -1) { // какая-то клетка была подсвечена
            ::SetWindowLongPtr(hWnd, GWLP_USERDATA, (LONG_PTR)-1);
            ::InvalidateRect(hWnd, NULL, FALSE);
        }
        // Передать дальше, чтобы система могла обрабатывать кнопки окна и рамку!
        return ::DefWindowProc(hWnd, message, wParam, lParam); // без неё не будет работать кнопка закрытия окна и рамка, т.к неизвестно, что мышь находится на рамке
    }

    case WM_MOUSELEAVE:


    case WM_PAINT:
    {
        PAINTSTRUCT paintstruct;
        HDC hDC = ::BeginPaint(hWnd, &paintstruct); // получаем контекст устройства для рисования

        RECT clientRect;
        ::GetClientRect(hWnd, &clientRect);

        int cols, rows;
        quantityOfRectangles(&cols, &rows);

        //Весь фон - белый
        HBRUSH hBgBrsh = (HBRUSH)::GetStockObject(WHITE_BRUSH); // стандартная кисть, удалять её не нужно
        ::FillRect(hDC, &clientRect, hBgBrsh);

        //Клетка под курсором, если курсор над какой-то клеткой
        int index = (int)::GetWindowLongPtr(hWnd, GWLP_USERDATA);
        if (index != -1) {
            RECT rcHover = GetCellRect(hWnd, index);
            HBRUSH hBrsh = ::CreateSolidBrush(RGB(229, 103, 23));

            //вот ето не есть хорошо
            ::FillRect(hDC, &rcHover, hBrsh);
            ::DeleteObject(hBrsh); // кисть создали сами, значит удаляем
        }

        // Линии сетки толщиной 2 пикселя - поверх заливки
        HPEN hPen = ::CreatePen(PS_SOLID, 2, RGB(0, 0, 0));
        HPEN oPen = (HPEN)::SelectObject(hDC, hPen); // запоминаем старое перо

        int cellW = clientRect.right / cols;
        int cellH = clientRect.bottom / rows;

        for (int c = 1; c < cols; ++c) { // вертикальные линии
            ::MoveToEx(hDC, c * cellW, 0, NULL);
            ::LineTo(hDC, c * cellW, clientRect.bottom);
        }

        for (int r = 1; r < rows; ++r) { // горизонтальные линии
            ::MoveToEx(hDC, 0, r * cellH, NULL);
            ::LineTo(hDC, clientRect.right, r * cellH);
        }

        // Вернуть старое перо и удалить своё
        ::SelectObject(hDC, oPen);
        ::DeleteObject(hPen);

        ::EndPaint(hWnd, &paintstruct); // закончили рисование
        return 0;
    }

    case WM_DESTROY:
        // Память мы не выделяли (данные лежат прямо в USERDATA), поэтому освобождать нечего.
        ::PostQuitMessage(0); // кладём в очередь WM_QUIT, цикл сообщений завершится
        return 0;
    }

    return ::DefWindowProc(hWnd, message, wParam, lParam);
}
